550:     // and the LEAST_FUNCTION_TYPE type at the bottom of the lattice.

552:     // When we introduce the unknown type, it's much more difficult to make
553:     // heads or tails of the partial ordering of types, because there's no
554:     // clear hierarchy between the different components (parameter types and
555:     // return types) in the ArrowType.
556:     //

559:     // approach of getting things right at the top and the bottom of the
560:     // lattice.
561:     if (isFunctionType() && that.isFunctionType()) {
562:       if (isEquivalentTo(that)) {
563:         return this;
564:       }

567:       if (that instanceof FunctionType) {
568:         other = (FunctionType) that;
569:       }
570: 
571:       // If these are ordinary functions, then merge them.
572:       // Don't do this if any of the params/return
573:       // values are unknown, because then there will be cycles in
574:       // their local lattice and they will merge in weird ways.
575:       if (other != null &&
576:           isOrdinaryFunction() && that.isOrdinaryFunction() &&
577:           !this.call.hasUnknownParamsOrReturn() &&
578:           !other.call.hasUnknownParamsOrReturn()) {
579: 
580:         // Check for the degenerate case, but double check
581:         // that there's not a cycle.
582:         boolean isSubtypeOfThat = this.isSubtype(that);
583:         boolean isSubtypeOfThis = that.isSubtype(this);
584:         if (isSubtypeOfThat && !isSubtypeOfThis) {
585:           return leastSuper ? that : this;
586:         } else if (isSubtypeOfThis && !isSubtypeOfThat) {
587:           return leastSuper ? this : that;
588:         }
589:

591:         FunctionType merged = tryMergeFunctionPiecewise(other, leastSuper);
592:         if (merged != null) {
593:           return merged;

598:       // that lives above the rest of the lattice.

600:           JSTypeNative.FUNCTION_INSTANCE_TYPE);
601:       if (functionInstance.isEquivalentTo(that)) {
602:         return leastSuper ? that : this;
603:       } else if (functionInstance.isEquivalentTo(this)) {
604:         return leastSuper ? this : that;
605:       }
