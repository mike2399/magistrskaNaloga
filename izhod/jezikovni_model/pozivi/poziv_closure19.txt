692:   private static Node computeFollowNode(
693:       Node fromNode, Node node, ControlFlowAnalysis cfa) {
694:     /*
695:      * This is the case where:
696:      *
697:      * 1. Parent is null implies that we are transferring control to the end of
698:      * the script.
699:      *
700:      * 2. Parent is a function implies that we are transferring control back to
701:      * the caller of the function.
702:      *
703:      * 3. If the node is a return statement, we should also transfer control
704:      * back to the caller of the function.
705:      *
706:      * 4. If the node is root then we have reached the end of what we have been
707:      * asked to traverse.
708:      *
709:      * In all cases we should transfer control to a "symbolic return" node.
710:      * This will make life easier for DFAs.
711:      */
712:     Node parent = node.getParent();
713:     if (parent == null || parent.isFunction() ||
714:         (cfa != null && node == cfa.root)) {
715:       return null;
716:     }
717: 
718:     // If we are just before a IF/WHILE/DO/FOR:
719:     switch (parent.getType()) {
720:       // The follow() of any of the path from IF would be what follows IF.
721:       case Token.IF:
722:         return computeFollowNode(fromNode, parent, cfa);
723:       case Token.CASE:
724:       case Token.DEFAULT_CASE:
725:         // After the body of a CASE, the control goes to the body of the next
726:         // case, without having to go to the case condition.
727:         if (parent.getNext() != null) {
728:           if (parent.getNext().isCase()) {
729:             return parent.getNext().getFirstChild().getNext();
730:           } else if (parent.getNext().isDefaultCase()) {
731:             return parent.getNext().getFirstChild();
732:           } else {
733:             Preconditions.checkState(false, "Not reachable");
734:           }
735:         } else {
736:           return computeFollowNode(fromNode, parent, cfa);
737:         }
738:         break;
739:       case Token.FOR:
740:         if (NodeUtil.isForIn(parent)) {
741:           return parent;
742:         } else {
743:           return parent.getFirstChild().getNext().getNext();
744:         }
745:       case Token.WHILE:
746:       case Token.DO:
747:         return parent;
748:       case Token.TRY:
749:         // If we are coming out of the TRY block...
750:         if (parent.getFirstChild() == node) {
751:           if (NodeUtil.hasFinally(parent)) { // and have FINALLY block.
752:             return computeFallThrough(parent.getLastChild());
753:           } else { // and have no FINALLY.
754:             return computeFollowNode(fromNode, parent, cfa);
