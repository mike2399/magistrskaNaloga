950:         // get the offset at instantLocal (first estimate)
951:         int offsetLocal = getOffset(instantLocal);
952:         // adjust instantLocal using the estimate and recalc the offset
953:         int offset = getOffset(instantLocal - offsetLocal);

958:             // value of offset is already correct for Eastern hemisphere)
959:             if (strict || offsetLocal < 0) {
960:                 // determine if we are in the DST gap
961:                 long nextLocal = nextTransition(instantLocal - offsetLocal);
962:                 if (nextLocal == (instantLocal - offsetLocal)) {
963:                     nextLocal = Long.MAX_VALUE;

965:                 long nextAdjusted = nextTransition(instantLocal - offset);
966:                 if (nextAdjusted == (instantLocal - offset)) {
