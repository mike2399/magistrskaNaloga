220:       // Make sure there is a space after e.g. instanceof , typeof
221:       append(" ");
222:     } else if (prev == '-' && first == '>') {
223:       // Make sure that we don't emit -->
224:       append(" ");
225:     }
226: 
227:     // Allow formating around the operator.
228:     appendOp(op, binOp);
229: 
230:     // Line breaking after an operator is always safe. Line breaking before an
231:     // operator on the other hand is not. We only line break after a bin op
232:     // because it looks strange.
233:     if (binOp) {
234:       maybeCutLine();
235:     }
236:   }
