683:       Node fromNode, Node node, ControlFlowAnalysis cfa) {
684:     /*
685:      * This is the case where:
686:      *
687:      * 1. Parent is null implies that we are transferring control to the end of
688:      * the script.
689:      *
690:      * 2. Parent is a function implies that we are transferring control back to
691:      * the caller of the function.
692:      *
693:      * 3. If the node is a return statement, we should also transfer control
694:      * back to the caller of the function.
695:      *
696:      * 4. If the node is root then we have reached the end of what we have been
697:      * asked to traverse.
698:      *
699:      * In all cases we should transfer control to a "symbolic return" node.
700:      * This will make life easier for DFAs.
701:      */
702:     Node parent = node.getParent();
703:     if (parent == null || parent.getType() == Token.FUNCTION ||
704:         (cfa != null && node == cfa.root)) {
705:       return null;
706:     }
707: 
708:     // If we are just before a IF/WHILE/DO/FOR:
709:     switch (parent.getType()) {
710:       // The follow() of any of the path from IF would be what follows IF.
711:       case Token.IF:
712:         return computeFollowNode(fromNode, parent, cfa);
713:       case Token.CASE:
714:       case Token.DEFAULT:

716:         // case, without having to go to the case condition.
717:         if (parent.getNext() != null) {
718:           if (parent.getNext().getType() == Token.CASE) {
719:             return parent.getNext().getFirstChild().getNext();
720:           } else if (parent.getNext().getType() == Token.DEFAULT) {
721:             return parent.getNext().getFirstChild();
722:           } else {
723:             Preconditions.checkState(false, "Not reachable");
724:           }
725:         } else {
726:           return computeFollowNode(fromNode, parent, cfa);
727:         }
728:         break;
729:       case Token.FOR:
730:         if (NodeUtil.isForIn(parent)) {
731:           return parent;
732:         } else {
733:           return parent.getFirstChild().getNext().getNext();
734:         }
735:       case Token.WHILE:
736:       case Token.DO:
737:         return parent;
738:       case Token.TRY:
739:         // If we are coming out of the TRY block...
740:         if (parent.getFirstChild() == node) {
741:           if (NodeUtil.hasFinally(parent)) { // and have FINALLY block.
742:             return computeFallThrough(parent.getLastChild());
743:           } else { // and have no FINALLY.
744:             return computeFollowNode(fromNode, parent, cfa);
745:           }
746:         // CATCH block.
747:         } else if (NodeUtil.getCatchBlock(parent) == node){
748:           if (NodeUtil.hasFinally(parent)) { // and have FINALLY block.
749:             return computeFallThrough(node.getNext());
750:           } else {
751:             return computeFollowNode(fromNode, parent, cfa);
752:           }
753:         // If we are coming out of the FINALLY block...
754:         } else if (parent.getLastChild() == node){
755:           if (cfa != null) {

757:               cfa.createEdge(fromNode, Branch.UNCOND, finallyNode);
758:             }
759:           }
760:           return computeFollowNode(fromNode, parent, cfa);
761:         }
762:     }

764:     // Now that we are done with the special cases follow should be its
765:     // immediate sibling, unless its sibling is a function
766:     Node nextSibling = node.getNext();

769:     while (nextSibling != null && nextSibling.getType() == Token.FUNCTION) {
