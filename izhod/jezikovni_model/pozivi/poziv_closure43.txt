728:       // See the comment in supAndInfHelper for more info on this.
729:       return thisType.toMaybeFunctionType().supAndInfHelper(
730:           thatType.toMaybeFunctionType(), false);
731:     } else if (thisType.isEquivalentTo(thatType)) {
732:       return thisType;
733:     } else if (thisType.isUnknownType() || thatType.isUnknownType()) {
734:       // The greatest subtype with any unknown type is the universal
735:       // unknown type, unless the two types are equal.
736:       return thisType.isEquivalentTo(thatType) ? thisType :
737:           thisType.getNativeType(JSTypeNative.UNKNOWN_TYPE);
738:     } else if (thisType.isSubtype(thatType)) {

741:       return filterNoResolvedType(thatType);
742:     } else if (thisType.isUnionType()) {
743:       return thisType.toMaybeUnionType().meet(thatType);
744:     } else if (thatType.isUnionType()) {
745:       return thatType.toMaybeUnionType().meet(thisType);
746:     } else if (thisType.isRecordType()) {
747:       return thisType.toMaybeRecordType().getGreatestSubtypeHelper(thatType);
748:     } else if (thatType.isRecordType()) {
749:       return thatType.toMaybeRecordType().getGreatestSubtypeHelper(thisType);
750:     }

752:     if (thisType.isEnumElementType()) {
753:       JSType inf = thisType.toMaybeEnumElementType().meet(thatType);
754:       if (inf != null) {
755:         return inf;
756:       }
757:     } else if (thatType.isEnumElementType()) {
