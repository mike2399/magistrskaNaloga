359:           // Determine if the string is a simple number.
360:           double d = getSimpleNumber(name);
361:           if (!Double.isNaN(d)) {
362:             cc.addNumber(d);
363:           } else {
364:             addJsString(n);
365:           }
366:         }
367: 
368:         add(parameters);
369:         add(body, Context.PRESERVE_BLOCK);
370:         break;
371: 
372:       case Token.SCRIPT:
373:       case Token.BLOCK: {
374:         if (n.getClass() != Node.class) {
375:           throw new Error("Unexpected Node subclass.");
376:         }
377:         boolean preserveBlock = context == Context.PRESERVE_BLOCK;
378:         if (preserveBlock) {
379:           cc.beginBlock();
380:         }
381: 
382:         boolean preferLineBreaks =
383:             type == Token.SCRIPT ||
384:             (type == Token.BLOCK &&
385:                 !preserveBlock &&
386:                 n.getParent() != null &&
387:                 n.getParent().isScript());
388:         for (Node c = first; c != null; c = c.getNext()) {
389:           add(c, Context.STATEMENT);
390: 
391:           // VAR doesn't include ';' since it gets used in expressions
392:           if (c.isVar()) {
393:             cc.endStatement();
394:           }
395: 
396:           if (c.isFunction()) {
397:             cc.maybeLineBreak();
398:           }
399: 
400:           // Prefer to break lines in between top-level statements
401:           // because top-level statements are more homogeneous.
402:           if (preferLineBreaks) {
403:             cc.notePreferredLineBreak();
404:           }
405:         }
406:         if (preserveBlock) {
407:           cc.endBlock(cc.breakAfterBlockFor(n, context == Context.STATEMENT));
408:         }
409:         break;
410:       }
411: 
412:       case Token.FOR:
413:         if (childCount == 4) {
414:           add("for(");
415:           if (first.isVar()) {
416:             add(first, Context.IN_FOR_INIT_CLAUSE);
417:           } else {
418:             addExpr(first, 0, Context.IN_FOR_INIT_CLAUSE);
419:           }
420:           add(";");
421:           add(first.getNext());
422:           add(";");
423:           add(first.getNext().getNext());
424:           add(")");
425:           addNonEmptyStatement(
426:               last, getContextForNonEmptyExpression(context), false);
427:         } else {
428:           Preconditions.checkState(childCount == 3);
429:           add("for(");
430:           add(first);
431:           add("in");
432:           add(first.getNext());
433:           add(")");
434:           addNonEmptyStatement(
435:               last, getContextForNonEmptyExpression(context), false);
436:         }
437:         break;
438: 
439:       case Token.DO:
440:         Preconditions.checkState(childCount == 2);
441:         add("do");
442:         addNonEmptyStatement(first, Context.OTHER, false);
443:         add("while(");
444:         add(last);
445:         add(")");
446:         cc.endStatement();
447:         break;
448: 
449:       case Token.WHILE:
450:         Preconditions.checkState(childCount == 2);
451:         add("while(");
452:         add(first);
453:         add(")");
454:         addNonEmptyStatement(
455:             last, getContextForNonEmptyExpression(context), false);
456:         break;
457: 
458:       case Token.EMPTY:
459:         Preconditions.checkState(childCount == 0);
460:         break;
461: 
462:       case Token.GETPROP: {
463:         Preconditions.checkState(
464:             childCount == 2,
465:             "Bad GETPROP: expected 2 children, but got %s", childCount);
466:         Preconditions.checkState(
467:             last.isString(),
468:             "Bad GETPROP: RHS should be STRING");
469:         boolean needsParens = (first.isNumber());
470:         if (needsParens) {
471:           add("(");
472:         }
473:         addExpr(first, NodeUtil.precedence(type), context);
474:         if (needsParens) {
475:           add(")");
476:         }
477:         add(".");
478:         addIdentifier(last.getString());
479:         break;
480:       }
481: 
482:       case Token.GETELEM:
483:         Preconditions.checkState(
484:             childCount == 2,
485:             "Bad GETELEM: expected 2 children but got %s", childCount);
486:         addExpr(first, NodeUtil.precedence(type), context);
487:         add("[");
488:         add(first.getNext());
489:         add("]");
490:         break;
491: 
492:       case Token.WITH:
493:         Preconditions.checkState(childCount == 2);
494:         add("with(");
495:         add(first);
496:         add(")");
497:         addNonEmptyStatement(
498:             last, getContextForNonEmptyExpression(context), false);
499:         break;
500: 
501:       case Token.INC:
502:       case Token.DEC: {
503:         Preconditions.checkState(childCount == 1);
504:         String o = type == Token.INC ? "++" : "--";
505:         int postProp = n.getIntProp(Node.INCRDECR_PROP);
506:         // A non-zero post-prop value indicates a post inc/dec, default of zero
507:         // is a pre-inc/dec.
508:         if (postProp != 0) {
509:           addExpr(first, NodeUtil.precedence(type), context);
510:           cc.addOp(o, false);
511:         } else {
512:           cc.addOp(o, false);
513:           add(first);
514:         }
515:         break;
516:       }
517: 
518:       case Token.CALL:
519:         // We have two special cases here:
520:         // 1) If the left hand side of the call is a direct reference to eval,
521:         // then it must have a DIRECT_EVAL annotation. If it does not, then
522:         // that means it was originally an indirect call to eval, and that
523:         // indirectness must be preserved.
524:         // 2) If the left hand side of the call is a property reference,
525:         // then the call must not a FREE_CALL annotation. If it does, then
526:         // that means it was originally an call without an explicit this and
527:         // that must be preserved.
528:         if (isIndirectEval(first)
529:             || n.getBooleanProp(Node.FREE_CALL) && NodeUtil.isGet(first)) {
530:           add("(0,");
531:           addExpr(first, NodeUtil.precedence(Token.COMMA), Context.OTHER);
532:           add(")");
533:         } else {
534:           addExpr(first, NodeUtil.precedence(type), context);
535:         }
536:         add("(");
537:         addList(first.getNext());
538:         add(")");
539:         break;
540: 
541:       case Token.IF:
542:         boolean hasElse = childCount == 3;
543:         boolean ambiguousElseClause =
544:             context == Context.BEFORE_DANGLING_ELSE && !hasElse;
545:         if (ambiguousElseClause) {
546:           cc.beginBlock();
547:         }
548: 
549:         add("if(");
550:         add(first);
551:         add(")");
552: 
553:         if (hasElse) {
554:           addNonEmptyStatement(
555:               first.getNext(), Context.BEFORE_DANGLING_ELSE, false);
556:           add("else");
557:           addNonEmptyStatement(
558:               last, getContextForNonEmptyExpression(context), false);
559:         } else {
560:           addNonEmptyStatement(first.getNext(), Context.OTHER, false);
561:           Preconditions.checkState(childCount == 2);
562:         }
563: 
564:         if (ambiguousElseClause) {
565:           cc.endBlock();
566:         }
567:         break;
568: 
569:       case Token.NULL:
570:         Preconditions.checkState(childCount == 0);
571:         cc.addConstant("null");
572:         break;
573: 
574:       case Token.THIS:
575:         Preconditions.checkState(childCount == 0);
576:         add("this");
577:         break;
578: 
579:       case Token.FALSE:
580:         Preconditions.checkState(childCount == 0);
581:         cc.addConstant("false");
582:         break;
583: 
584:       case Token.TRUE:
585:         Preconditions.checkState(childCount == 0);
586:         cc.addConstant("true");
587:         break;
588: 
589:       case Token.CONTINUE:
590:         Preconditions.checkState(childCount <= 1);
591:         add("continue");
592:         if (childCount == 1) {
593:           if (!first.isLabelName()) {
594:             throw new Error("Unexpected token type. Should be LABEL_NAME.");
595:           }
596:           add(" ");
597:           add(first);
598:         }
599:         cc.endStatement();
600:         break;
601: 
602:       case Token.DEBUGGER:
603:         Preconditions.checkState(childCount == 0);
604:         add("debugger");
605:         cc.endStatement();
606:         break;
607: 
608:       case Token.BREAK:
609:         Preconditions.checkState(childCount <= 1);
610:         add("break");
611:         if (childCount == 1) {
612:           if (!first.isLabelName()) {
613:             throw new Error("Unexpected token type. Should be LABEL_NAME.");
614:           }
615:           add(" ");
616:           add(first);
617:         }
618:         cc.endStatement();
619:         break;
620: 
621:       case Token.EXPR_RESULT:
622:         Preconditions.checkState(childCount == 1);
623:         add(first, Context.START_OF_EXPR);
624:         cc.endStatement();
625:         break;
626: 
627:       case Token.NEW:
628:         add("new ");
629:         int precedence = NodeUtil.precedence(type);
630: 
631:         // If the first child contains a CALL, then claim higher precedence
632:         // to force parentheses. Otherwise, when parsed, NEW will bind to the
633:         // first viable parentheses (don't traverse into functions).
634:         if (NodeUtil.containsType(
635:             first, Token.CALL, NodeUtil.MATCH_NOT_FUNCTION)) {
636:           precedence = NodeUtil.precedence(first.getType()) + 1;
637:         }
638:         addExpr(first, precedence, Context.OTHER);
639: 
640:         // '()' is optional when no arguments are present
641:         Node next = first.getNext();
642:         if (next != null) {
643:           add("(");
644:           addList(next);
645:           add(")");
646:         }
647:         break;
648: 
649:       case Token.STRING_KEY:
650:         Preconditions.checkState(
651:             childCount == 1, "Object lit key must have 1 child");
652:         addJsString(n);
653:         break;
654: 
655:       case Token.STRING:
656:         Preconditions.checkState(
657:             childCount == 0, "A string may not have children");
658:         addJsString(n);
659:         break;
660: 
661:       case Token.DELPROP:
662:         Preconditions.checkState(childCount == 1);
663:         add("delete ");
664:         add(first);
665:         break;
666: 
667:       case Token.OBJECTLIT: {
668:         boolean needsParens = (context == Context.START_OF_EXPR);
669:         if (needsParens) {
670:           add("(");
671:         }
672:         add("{");
673:         for (Node c = first; c != null; c = c.getNext()) {
674:           if (c != first) {
675:             cc.listSeparator();
676:           }
677: 
678:           if (c.isGetterDef() || c.isSetterDef()) {
679:             add(c);
680:           } else {
681:             Preconditions.checkState(c.isStringKey());
682:             String key = c.getString();
683:             // Object literal property names don't have to be quoted if they
684:             // are not JavaScript keywords
685:             if (!c.isQuotedString() &&
686:                 !TokenStream.isKeyword(key) &&
687:                 TokenStream.isJSIdentifier(key) &&
688:                 // do not encode literally any non-literal characters that
689:                 // were Unicode escaped.
690:                 NodeUtil.isLatin(key)) {
691:               add(key);
692:             } else {
693:               // Determine if the string is a simple number.
694:               double d = getSimpleNumber(key);
695:               if (!Double.isNaN(d)) {
696:                 cc.addNumber(d);
697:               } else {
698:                 addExpr(c, 1, Context.OTHER);
699:               }
700:             }
701:             add(":");
702:             addExpr(c.getFirstChild(), 1, Context.OTHER);
703:           }
704:         }
705:         add("}");
706:         if (needsParens) {
707:           add(")");
708:         }
709:         break;
710:       }
711: 
712:       case Token.SWITCH:
713:         add("switch(");
714:         add(first);
715:         add(")");
716:         cc.beginBlock();
717:         addAllSiblings(first.getNext());
718:         cc.endBlock(context == Context.STATEMENT);
719:         break;
720: 
721:       case Token.CASE:
722:         Preconditions.checkState(childCount == 2);
723:         add("case ");
724:         add(first);
725:         addCaseBody(last);
726:         break;
727: 
728:       case Token.DEFAULT_CASE:
729:         Preconditions.checkState(childCount == 1);
730:         add("default");
731:         addCaseBody(first);
732:         break;
733: 
734:       case Token.LABEL:
735:         Preconditions.checkState(childCount == 2);
736:         if (!first.isLabelName()) {
737:           throw new Error("Unexpected token type. Should be LABEL_NAME.");
738:         }
739:         add(first);
740:         add(":");
741:         addNonEmptyStatement(
742:             last, getContextForNonEmptyExpression(context), true);
743:         break;
744: 
745:       case Token.CAST:
746:         add("(");
747:         add(first);
748:         add(")");
749:         break;
750: 
751:       default:
752:         throw new Error("Unknown type " + type + "\n" + n.toStringTree());
753:     }
754: 
755:     cc.endSourceMapping(n);
756:   }
