1862:       // {?=} - equals

1864:       // {function(number, ?)} - right paren
1865:       // {function(number, ...[?])} - right bracket
1866:       // {function(): ?|number} - pipe
1867:       // {Array.<?>} - greater than
1868:       // /** ? */ - EOC (inline types)

1870:       // the type language a lot nicer.

1874:           token == JsDocToken.RB ||
1875:           token == JsDocToken.RC ||

1877:           token == JsDocToken.PIPE ||
1878:           token == JsDocToken.GT ||
1879:           token == JsDocToken.EOC) {

1881:         return newNode(Token.QMARK);
