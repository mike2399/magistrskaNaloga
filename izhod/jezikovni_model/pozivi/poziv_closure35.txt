229:   static boolean isSubtype(ObjectType typeA, RecordType typeB) {
230:     // typeA is a subtype of record type typeB iff:
231:     // 1) typeA has all the properties declared in typeB.
232:     // 2) And for each property of typeB,
233:     //    2a) if the property of typeA is declared, it must be equal
234:     //        to the type of the property of typeB,
235:     //    2b) otherwise, it must be a subtype of the property of typeB.
236:     //
237:     // To figure out why this is true, consider the following pseudo-code:
238:     // /** @type {{a: (Object,null)}} */ var x;
239:     // /** @type {{a: !Object}} */ var y;
240:     // var z = {a: {}};
241:     // x.a = null;
242:     //
243:     // y cannot be assigned to x, because line 4 would violate y's declared
244:     // properties. But z can be assigned to x. Even though z and y are the
245:     // same type, the properties of z are inferred--and so an assignment
246:     // to the property of z would not violate any restrictions on it.
247:     for (String property : typeB.properties.keySet()) {
248:       if (!typeA.hasProperty(property)) {
249:         return false;
250:       }
251: 
252:       JSType propA = typeA.getPropertyType(property);
253:       JSType propB = typeB.getPropertyType(property);
254:       if (!propA.isUnknownType() && !propB.isUnknownType()) {
255:         if (typeA.isPropertyTypeDeclared(property)) {
256:           if (!propA.isEquivalentTo(propB)) {
257:             return false;
258:           }
259:         } else {
260:           if (!propA.isSubtype(propB)) {
261:             return false;

263:         }
