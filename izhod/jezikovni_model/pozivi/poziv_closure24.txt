350:             return;
351:           } else {
352:             report(t, n, GOOG_SCOPE_ALIAS_REDEFINED, n.getString());
353:           }
354:         }
355: 
356:         if (type == Token.RETURN) {
357:           report(t, n, GOOG_SCOPE_USES_RETURN);
358:         } else if (type == Token.THIS) {
359:           report(t, n, GOOG_SCOPE_REFERENCES_THIS);
360:         } else if (type == Token.THROW) {
361:           report(t, n, GOOG_SCOPE_USES_THROW);
362:         }
363:       }
364: 
365:       // Validate all descendent scopes of the goog.scope block.
366:       if (t.getScopeDepth() >= 2) {
367:         // Check if this name points to an alias.
368:         if (aliasVar != null) {
369:           // Note, to support the transitive case, it's important we don't
370:           // clone aliasedNode here.  For example,
371:           // var g = goog; var d = g.dom; d.createElement('DIV');
372:           // The node in aliasedNode (which is "g") will be replaced in the
373:           // changes pass above with "goog".  If we cloned here, we'd end up
374:           // with <code>g.dom.createElement('DIV')</code>.
375:           Node aliasedNode = aliasVar.getInitialValue();
376:           aliasUsages.add(new AliasedNode(n, aliasedNode));
377:         }
378: 
379:         JSDocInfo info = n.getJSDocInfo();
380:         if (info != null) {
381:           for (Node node : info.getTypeNodes()) {
382:             fixTypeNode(node);
383:           }
384:         }
385: 
386:         // TODO(robbyw): Error for goog.scope not at root.
387:       }
388:     }
